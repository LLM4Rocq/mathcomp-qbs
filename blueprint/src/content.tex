\chapter*{Introduction}

Quasi-Borel spaces solve a fundamental problem: the category of
measurable spaces is not cartesian closed, which prevents giving
semantics to higher-order probabilistic programs. QBS replaces
measurable sets with ``random elements'' (morphisms from~$\mathbb{R}$),
yielding a cartesian-closed category with a well-behaved probability
monad.

This formalization follows:
\begin{itemize}
\item C.~Heunen, O.~Kammar, S.~Staton, H.~Yang.
  \emph{A Convenient Category for Higher-Order Probability Theory.}
  LICS 2017.
  \url{https://arxiv.org/abs/1701.02547}
\item M.~Hirata, Y.~Minamide, T.~Sato.
  \emph{Quasi-Borel Spaces.}
  Archive of Formal Proofs, 2022.
  \url{https://www.isa-afp.org/entries/Quasi_Borel_Spaces.html}
\end{itemize}

\chapter{Core QBS Theory}\label{ch:core}

In this chapter we define quasi-Borel spaces, their morphisms, and
the fundamental categorical constructions: products, exponentials,
coproducts, dependent products, and the unit type.
We establish cartesian closure, the $L \dashv R$ adjunction between
QBS and measurable spaces, and introduce standard Borel spaces
with their full faithfulness property.

The development is parametric in a \texttt{realType}~$R$ from math-comp
analysis. We write $\mathbb{R}$ for the carrier of~$R$ and
$m\mathbb{R}$ for its Borel measurable space.

\section{QBS Definition}

\begin{definition}[Quasi-Borel Space]\label{def:isQBS}
  \rocq{QBS.quasi_borel.isQBS}
  \rocqok
  A \emph{quasi-Borel space} (QBS) over $R$ is a type~$T$ equipped with
  a set $M_X \subseteq (\mathbb{R} \to T)$ of \emph{random elements}
  satisfying three axioms:
  \begin{enumerate}
    \item \textbf{Composition} (\texttt{qbs\_Mx\_comp}):
      If $\alpha \in M_X$ and $f : \texttt{\{mfun mR >-> mR\}}$
      (a bundled measurable function), then
      $\alpha \circ f \in M_X$.
    \item \textbf{Constants} (\texttt{qbs\_Mx\_const}):
      For every $x : T$, the constant function
      $\lambda r.\; x$ is in $M_X$.
    \item \textbf{Gluing} (\texttt{qbs\_Mx\_glue}):
      If $P : \texttt{\{mfun mR >-> nat\}}$ and
      $F_i \in M_X$ for all~$i$, then
      $\lambda r.\; F_{P(r)}(r) \in M_X$.
  \end{enumerate}
  The HB structure \texttt{QBSpace} packages this mixin with a short
  type name \texttt{qbsType R}. All type constructions
  (\texttt{prodQ}, \texttt{expQ}, \texttt{unitQ}, etc.) use the
  Section + \texttt{HB.instance Definition} pattern recommended by
  mathcomp maintainers.
\end{definition}


\section{Morphisms}\label{sec:morphisms}

\begin{definition}[QBS Morphism]\label{def:qbs_morphism}
  \rocq{QBS.quasi_borel.qbs_morphism}
  \rocqok
  \uses{def:isQBS}
  A function $f : X \to Y$ between QBS is a \emph{QBS morphism} if for
  every $\alpha \in M_X$, we have $f \circ \alpha \in M_Y$.
\end{definition}

\begin{lemma}[Identity Morphism]\label{lem:qbs_morphism_id}
  \rocq{QBS.quasi_borel.qbs_id}
  \rocqok
  \uses{def:qbs_morphism}
  The identity function $\mathrm{id} : X \to X$ is a QBS morphism.
\end{lemma}
\begin{proof}\rocqok\end{proof}

\begin{lemma}[Composition of Morphisms]\label{lem:qbs_morphism_comp}
  \rocq{QBS.quasi_borel.qbs_comp}
  \rocqok
  \uses{def:qbs_morphism}
  If $f : X \to Y$ and $g : Y \to Z$ are QBS morphisms, then
  $g \circ f : X \to Z$ is a QBS morphism.
\end{lemma}
\begin{proof}\rocqok\end{proof}

\begin{lemma}[Constant Morphism]\label{lem:qbs_morphism_const}
  \rocq{QBS.quasi_borel.qbs_const}
  \rocqok
  \uses{def:qbs_morphism}
  For any $y : Y$, the constant function $\lambda x.\; y$ is a QBS morphism.
\end{lemma}
\begin{proof}\rocqok\end{proof}

\begin{lemma}[Category Laws]\label{lem:qbs_cat_laws}
  \rocq{QBS.quasi_borel.qbs_homeqP}
  \rocq{QBS.quasi_borel.qbs_1comp}
  \rocq{QBS.quasi_borel.qbs_comp1}
  \rocq{QBS.quasi_borel.qbs_compA}
  \rocqok
  \uses{lem:qbs_morphism_id, lem:qbs_morphism_comp}
  Two bundled QBS morphisms are equal iff they agree pointwise
  (the proof component is a proposition).
  Consequently identities are units and composition is associative,
  as \emph{equalities} of bundled morphisms: $\mathrm{id} \circ f = f$,
  $f \circ \mathrm{id} = f$, and $h \circ (g \circ f) = (h \circ g) \circ f$.
  QBS and QBS morphisms form a category.
\end{lemma}
\begin{proof}\rocqok\end{proof}

\section{The R Functor}\label{sec:R_functor}

  are measurable ($T$ any measurable space), then $\lambda r.\; F_{P(r)}(r)$ is measurable.
  This justifies the gluing axiom for~$R(M)$.
\end{lemma}
\begin{proof}\rocqok\end{proof}

  $\mathtt{boolQ} = R(\mathbb{B})$.
\end{definition}

\begin{lemma}[R Preserves Morphisms]\label{lem:R_qbs_morph}
  \rocq{QBS.measure_qbs_adjunction.measQ_morphism}
  \rocqok

\section{Exponentials (Function Spaces)}\label{sec:exponentials}

\begin{definition}[Exponential QBS]\label{def:expQ}
  \rocq{QBS.quasi_borel.expQ}
  \rocqok
  \uses{def:isQBS, def:qbs_morphism}
  The exponential $X \Rightarrow Y$ has carrier the set of QBS morphisms
  $X \to Y$, with a function $g : \mathbb{R} \to (X \to Y)$ random iff the
  uncurried map $(r, x) \mapsto g(r)(x)$ is a QBS morphism
  $\mathbb{R} \times X \to Y$.
\end{definition}

\begin{theorem}[Evaluation Morphism]\label{thm:qbs_morphism_eval}
  \rocq{QBS.quasi_borel.qbs_eval}
  \rocqok
  \uses{def:expQ, def:prodQ, def:qbs_morphism}
  Evaluation $(f, x) \mapsto f(x)$ is a QBS morphism
  $(X \Rightarrow Y) \times X \to Y$.
\end{theorem}
\begin{proof}\rocqok\end{proof}

\begin{theorem}[Currying Morphism]\label{thm:qbs_morphism_curry}
  \rocq{QBS.quasi_borel.qbs_curry_morph}
  \rocqok
  \uses{def:expQ, def:prodQ, def:qbs_morphism, lem:qbs_Mx_pair_cstl}
  If $f : X \times Y \to Z$ is a QBS morphism, then
  $\hat{f} : X \to (Y \Rightarrow Z)$ defined by
  $\hat{f}(x)(y) = f(x,y)$ is a QBS morphism
  (for each fixed $x$, \texttt{qbs\_curry}$\;f\;x : Y \to Z$ is itself a
  morphism).
\end{theorem}
\begin{proof}\rocqok\end{proof}

\begin{definition}[Uncurrying]\label{def:qbs_uncurry}
  \rocq{QBS.quasi_borel.qbs_uncurry}
  \rocqok
  \uses{thm:qbs_morphism_eval, thm:qbs_morphism_curry, lem:qbs_morphism_pair, lem:qbs_morphism_fst, lem:qbs_morphism_snd}
  For a QBS morphism $g : X \to (Y \Rightarrow Z)$, the uncurried morphism
  $X \times Y \to Z$ is $\mathrm{eval} \circ (g \circ \pi_1, \pi_2)$.
\end{definition}

\begin{theorem}[Universal Property of the Exponential]\label{thm:qbs_curry_universal}
  \rocq{QBS.quasi_borel.qbs_curryK}
  \rocq{QBS.quasi_borel.qbs_uncurryK}
  \rocq{QBS.quasi_borel.qbs_curry_morph_uniq}
  \rocqok
  \uses{thm:qbs_morphism_curry, def:qbs_uncurry, lem:qbs_cat_laws}
  Currying and uncurrying are inverse bijections between
  $\mathrm{Hom}(X \times Y, Z)$ and $\mathrm{Hom}(X, Y \Rightarrow Z)$, as
  equalities of bundled morphisms: $\mathrm{uncurry}(\mathrm{curry}\,f) = f$
  ($\beta$), $\mathrm{curry}(\mathrm{uncurry}\,g) = g$ ($\eta$), and the
  transpose is unique: $\mathrm{uncurry}\, g = f$ implies
  $g = \mathrm{curry}\, f$.
\end{theorem}
\begin{proof}\rocqok\end{proof}

\begin{lemma}[Internal Currying]\label{lem:qbs_curry_exp}
  \rocq{QBS.quasi_borel.qbs_curry_exp}
  \rocq{QBS.quasi_borel.qbs_uncurry_exp}
  \rocqok
  \uses{thm:qbs_curry_universal, def:expQ}
  The bijection is internal: currying and uncurrying are themselves QBS
  morphisms $((X \times Y) \Rightarrow Z) \to (X \Rightarrow (Y \Rightarrow Z))$
  and back.
\end{lemma}
\begin{proof}\rocqok\end{proof}

\begin{corollary}[Cartesian Closure]\label{cor:cartesian_closure}
  \rocqok
  \uses{thm:qbs_morphism_eval, thm:qbs_curry_universal, lem:qbs_morphism_unit}
  The category of quasi-Borel spaces is cartesian closed
  (HKSY Prop.~16): it has the terminal object \texttt{unitQ},
  binary products, and exponentials with the universal property above.
\end{corollary}
\begin{proof}\rocqok\end{proof}

\begin{lemma}[Exponential Composition]\label{lem:qbs_morphism_exp_comp}
  \rocq{QBS.quasi_borel.qbs_exp_comp}
  \rocqok
  \uses{def:expQ, def:prodQ, def:qbs_morphism}
  Given QBS morphisms $f : W \to (X \Rightarrow Y)$ and $g : W \to X$,
  the pointwise application $w \mapsto f(w)(g(w))$ is a QBS morphism
  $W \to Y$.
\end{lemma}
\begin{proof}\rocqok\end{proof}

  \rocq{QBS.quasi_borel.qbs_closed_Mx}

\begin{proof}\rocqok\end{proof}

\begin{lemma}[Top, Bottom and Empty: Properties]\label{lem:qbs_topT_props}
  \rocq{QBS.quasi_borel.qbs_top_ub}
  \rocq{QBS.quasi_borel.qbs_bot_lb}

  $f$ is a QBS morphism $X \to R(M)$ if and only if $f$ is a measurable

  \[
    \mathrm{Hom}_{\mathbf{QBS}}(X, R(M)) \cong
    \mathrm{Hom}_{\mathbf{Meas}}(L(X), M).

  measurable $\alpha : \mathbb{R} \to L(X)$ with the measurable $f$.
\end{proof}

  \rocqok

  standard Borel domains.
\end{theorem}
\begin{proof}
  \rocqok

  of $R(M_1)$, hence $f \circ g$ is measurable, and $e$ is measurable.

  \uses{def:isQBS}

\chapter{Probability Monad}\label{ch:probability}

This chapter develops the QBS probability monad $P(X)$: the type of
probability distributions on a QBS~$X$, following HKSY Sections~IV--V.
A probability is a pair of a random element and a probability measure on
$\mathbb{R}$, taken up to equivalence; the random elements of $P(X)$ are
the families $r \mapsto [\alpha, g(r)]$ with $g : \mathbb{R} \to
\mathrm{Giry}(\mathbb{R})$ measurable, and bind is the Giry bind of
math-comp analysis.
We prove the monad laws, the strength, internal bind and commutativity
(HKSY Thm.~21 and Prop.~22), develop integration, normalisation
and the quotient by equivalence, and establish independence
and variance results.

\begin{lemma}[Centred First Moment of the Normal Density]\label{lem:integral_normal_centred}
  \rocq{QBS.normal_centred.integral_normal_centred}
  \rocqok
  For $\sigma \neq 0$, the function
  $x \mapsto (x - \mu)\,\mathrm{pdf}_{\mu,\sigma}(x)$ is Lebesgue integrable
  and $\int (x - \mu)\,\mathrm{pdf}_{\mu,\sigma}(x)\,dx = 0$.
\end{lemma}
\begin{proof}\rocqok
  On $[\mu, +\infty)$ the integral is $\sigma^2 \cdot \mathrm{pdf}_{\mu,\sigma}(\mu)$
  by the fundamental theorem of calculus, with antiderivative
  $-\sigma^2\,\mathrm{pdf}_{\mu,\sigma}$; on $(-\infty, \mu]$ it is the
  opposite, by the substitution $x \mapsto -x$.
\end{proof}

\begin{lemma}[First Absolute Moment of the Normal]\label{lem:normal_prob_abs_lty}
  \rocq{QBS.normal_centred.normal_prob_abs_lty}
  \rocqok
  \uses{lem:integral_normal_centred}
  For nonnegative measurable $f$,
  $\int f\, d\mathcal{N}(\mu,\sigma) = \int f(x)\,\mathrm{pdf}_{\mu,\sigma}(x)\,dx$.
  Consequently, for $\sigma \neq 0$,
  $\int |x|\, d\mathcal{N}(\mu,\sigma) < +\infty$.
\end{lemma}
\begin{proof}\rocqok
  Radon--Nikodym change of variables, then
  $|x| \le |x - \mu| + |\mu|$.
\end{proof}

\begin{lemma}[Mean of the Normal Law]\label{lem:integral_normal_mean}
  \rocq{QBS.normal_centred.integral_normal_mean}
  \rocqok
  \uses{lem:integral_normal_centred, lem:normal_prob_abs_lty}
  For $\sigma \neq 0$, $\int x\, d\mathcal{N}(\mu,\sigma)(x) = \mu$.
\end{lemma}
\begin{proof}\rocqok
  Split $x = (x-\mu) + \mu$ and use the centred moment lemma.
\end{proof}

  \rocqok

\begin{proof}\rocqok\end{proof}

\chapter{Standard Borel Spaces ($\mathbb{R} \cong \mathbb{R} \times \mathbb{R}$)}\label{ch:standard_borel}

This chapter constructs the measurable bijection
$\mathbb{R} \cong \mathbb{R} \times \mathbb{R}$, proving that
the product of standard Borel spaces is standard Borel.
The construction composes three layers: (1) a bijection
$\mathbb{R} \leftrightarrow (0,1)$ via arctangent, (2) a bijection
$(0,1)^2 \leftrightarrow (0,1)$ via binary digit interleaving,
and (3) the composed encode/decode pair.

  $\mathrm{bin\_digit}(x, n)$ extracts the $n$-th binary digit of
  $x \in [0,1)$ via the doubling map
  $\mathrm{bin\_step}(y) = 2y - \lfloor 2y \rfloor$ (\texttt{bin\_rem x n} is its $n$-th iterate, with
  $\mathrm{bin\_digit}(x,n) = (1/2 \le \mathrm{bin\_rem}(x,n))$ and

\end{definition}

  \rocqok
  \uses{def:bin_digits}
  Partial sums converge, are non-decreasing, and the limit satisfies
  $0 \leq \mathrm{bin\_sum}(d) \leq 1$. The maps \texttt{bin\_step}, \texttt{bin\_rem} and \texttt{bin\_sum} are measurable.
\end{lemma}
\begin{proof}\rocqok\end{proof}

\end{lemma}

  \uses{thm:encode_RRK, lem:measurable_encode_decode}
  Take $f = \mathrm{encode\_RR}$, $g = \mathrm{decode\_RR}$.
  Measurability is by \texttt{measurable\_encode\_RR} and

\chapter{Bayesian Linear Regression}\label{ch:bayesian}

This chapter formalizes the Bayesian linear regression example of HKSY
Sec.~VI, following the Isabelle AFP development by Hirata, Minamide, and
Sato (file \texttt{showcase/bayesian\_regression.v}).
The model is $y = s \cdot x + b + \varepsilon$ with
$\varepsilon \sim \mathcal{N}(0, 1/2)$,
prior $s, b \sim \mathrm{iid}\; \mathcal{N}(0, 3)$,
and data $(1,2.5), (2,3.8), (3,4.5), (4,6.2), (5,8.0)$.
The prior is a probability on the function space $\mathbb{R} \Rightarrow
\mathbb{R}$, the program is $\mathrm{norm}(P(\mathrm{id}, \mathrm{obs})
(\mathrm{prior}))$, and the posterior is computed in closed form
by completing the square.
All algebra is done inside the file with \texttt{field}/\texttt{ring} from
math-comp algebra-tactics.

\section{Model}\label{sec:priors}

\begin{definition}[Parameter Prior]\label{def:prior2}
  \rocq{QBS.showcase.bayesian_regression.nu}
  \rocq{QBS.showcase.bayesian_regression.prior2}
  \rocqok
  \uses{def:qbs_normal_distribution, def:monadP_pair}
  $\nu = \mathcal{N}(0, 3)$ and the parameter prior
  $\mathrm{prior2} = \nu \otimes \nu$ on $\mathbb{R} \times \mathbb{R}$,
  a monadic product.
\end{definition}

\begin{definition}[Linear Functions]\label{def:lin}
  \rocq{QBS.showcase.bayesian_regression.lin_uncurried}
  \rocq{QBS.showcase.bayesian_regression.lin}
  \rocq{QBS.showcase.bayesian_regression.linE}
  \rocqok
  \uses{thm:qbs_morphism_curry, lem:qbs_morphism_add, lem:qbs_morphism_mul}
  $\mathrm{lin} : \mathbb{R} \times \mathbb{R} \to (\mathbb{R} \Rightarrow
  \mathbb{R})$ is the curried QBS morphism with
  $\mathrm{lin}(s, b)(r) = s \cdot r + b$.
\end{definition}

\begin{definition}[Prior on Functions]\label{def:prior}
  \rocq{QBS.showcase.bayesian_regression.prior}
  \rocqok
  \uses{def:prior2, def:lin, def:monadP_map}
  $\mathrm{prior} = P(\mathrm{lin})(\mathrm{prior2})$, a probability on
  $\mathbb{R} \Rightarrow \mathbb{R}$ (HKSY eq.~(5)).
\end{definition}

\begin{definition}[Observation Likelihood]\label{def:obs}
  \rocq{QBS.showcase.bayesian_regression.obs_fun}
  \rocq{QBS.showcase.bayesian_regression.obs_morphism}
  \rocq{QBS.showcase.bayesian_regression.obs}
  \rocqok
  \uses{def:expQ, def:concrete_qbs}
  For a regression function $f : \mathbb{R} \Rightarrow \mathbb{R}$,
  \[
    \mathrm{obs}(f) = \prod_{k=1}^{5}
    \mathcal{N}(f(k),\; \tfrac{1}{2})(y_k)
  \]
  (the normal density at $y_k$) where
  $(y_1, \ldots, y_5) = (2.5, 3.8, 4.5, 6.2, 8.0)$. It is a QBS
  morphism $(\mathbb{R} \Rightarrow \mathbb{R}) \to \mathtt{realQ}$.
\end{definition}

\begin{definition}[Weighted Prior and Program]\label{def:program}
  \rocq{QBS.showcase.bayesian_regression.wprior}
  \rocq{QBS.showcase.bayesian_regression.program}
  \rocqok
  \uses{def:prior, def:obs, def:qbs_normalize, def:monadP_map}
  $\mathrm{wprior} = P(\mathrm{id}, \mathrm{obs})(\mathrm{prior})$ is a
  probability on $(\mathbb{R} \Rightarrow \mathbb{R}) \times
  \mathbb{R}$, and $\mathrm{program} = \mathrm{norm}(\mathrm{wprior})$
  is an option of a probability on $\mathbb{R} \Rightarrow \mathbb{R}$.
\end{definition}


\section{Closed-Form Posterior}\label{sec:posterior}

\begin{definition}[Posterior Parameters and Evidence Constant]\label{def:post_params}
  \rocq{QBS.showcase.bayesian_regression.evidence_const}
  \rocq{QBS.showcase.bayesian_regression.post_s_mean}
  \rocq{QBS.showcase.bayesian_regression.post_s_var}
  \rocq{QBS.showcase.bayesian_regression.post_b_mean}
  \rocq{QBS.showcase.bayesian_regression.post_b_var}
  \rocqok
  Explicit rational constants for the posterior: the slope has mean
  $m_s = \frac{8\cdot 9\cdot 7\cdot 19\cdot 47}{5\cdot 29\cdot 2309}$ and
  variance $v_s = \frac{9 \cdot 181}{29 \cdot 2309}$; given the slope~$s$,
  the intercept has mean $m_b(s) = \frac{900 - 540 s}{181}$ and variance
  $v_b = \frac{9}{181}$; $C$ (\texttt{evidence\_const}) is the closed-form evidence constant
  \[
    C = \frac{4\sqrt{2}}{\pi^2 \sqrt{29 \cdot 2309 \cdot \pi}}
    \exp\Bigl(-\frac{11 \cdot 11 \cdot (138\cdot 100 + 41)}
    {25 \cdot 29 \cdot 2309}\Bigr).
  \]
\end{definition}

\begin{definition}[Posterior]\label{def:posterior}
  \rocq{QBS.showcase.bayesian_regression.post_b_meanQ}
  \rocq{QBS.showcase.bayesian_regression.posterior2}
  \rocq{QBS.showcase.bayesian_regression.posterior}
  \rocqok
  \uses{def:post_params, def:qbs_bind, def:qbs_strength, def:lin, lem:qbs_normal_morphism}
  $\mathrm{posterior2}$ draws $s \sim \mathcal{N}(m_s, \sqrt{v_s})$ and
  then $b \sim \mathcal{N}(m_b(s), \sqrt{v_b})$, built with bind and
  strength; $\mathrm{posterior} = P(\mathrm{lin})(\mathrm{posterior2})$.
\end{definition}

\begin{lemma}[Completing the Square]\label{lem:obs_prior_factor}
  \rocq{QBS.showcase.bayesian_regression.obs_prior_factor}
  \rocqok
  \uses{def:obs, def:post_params, def:prior2}
  Prior times likelihood is $C$ times the posterior density:
  \[
    \mathcal{N}(0,3)(s)\cdot\mathcal{N}(0,3)(b)\cdot
    \mathrm{obs}(\mathrm{lin}(s,b))
    = C \cdot \mathcal{N}(m_s, \sqrt{v_s})(s)
    \cdot \mathcal{N}(m_b(s), \sqrt{v_b})(b).
  \]
\end{lemma}
\begin{proof}\rocqok\end{proof}

\begin{lemma}[Integrals Against Prior and Posterior]\label{lem:prior_posterior_integral}
  \rocq{QBS.showcase.bayesian_regression.prior2_integral}
  \rocq{QBS.showcase.bayesian_regression.posterior2_integral}
  \rocqok
  \uses{def:prior2, def:posterior, lem:qbs_bind_event}
  For QBS-measurable $h \geq 0$, $\int_{\mathrm{prior2}} h =
  \int_{\mathcal{N}(0,3)}\int_{\mathcal{N}(0,3)} h(s,b)$ and
  $\int_{\mathrm{posterior2}} h = \int_{\mathcal{N}(m_s,\sqrt{v_s})}
  \int_{\mathcal{N}(m_b(s),\sqrt{v_b})} h(s,b)$.
\end{lemma}
\begin{proof}\rocqok\end{proof}

\begin{lemma}[Factorisation of the Weighted Integral]\label{lem:factor_integral}
  \rocq{QBS.showcase.bayesian_regression.factor_integral}
  \rocqok
  \uses{lem:obs_prior_factor, lem:prior_posterior_integral}
  For QBS-measurable $h$ with $0 \le h \le 1$:
  $\int_{\mathcal{N}(0,3)}\int_{\mathcal{N}(0,3)} h(s,b)\,
  \mathrm{obs}(\mathrm{lin}(s,b)) = C \cdot
  \int_{\mathcal{N}(m_s,\sqrt{v_s})}\int_{\mathcal{N}(m_b(s),\sqrt{v_b})}
  h(s,b)$.
\end{lemma}
\begin{proof}
  \rocqok
  \uses{lem:obs_prior_factor}
  Convert the integrals against normal probabilities into Lebesgue
  integrals with density (the integrands are bounded), apply
  Lemma~\ref{lem:obs_prior_factor} pointwise, and use measurability of the
  inner integrals.
\end{proof}


\section{Main Results}\label{sec:main_results}

\begin{theorem}[Evidence of the Weighted Prior]\label{thm:wprior_evidence}
  \rocq{QBS.showcase.bayesian_regression.wprior_evidence}
  \rocqok
  \uses{def:program, def:qbs_evidence, lem:factor_integral, lem:prior_posterior_integral}
  The evidence of $\mathrm{wprior}$ is the closed-form constant:
  $Z_{\mathrm{wprior}} = C$. In particular $0 < Z < +\infty$.
\end{theorem}
\begin{proof}\rocqok\end{proof}

\begin{theorem}[Main Theorem: the Program Computes the Posterior]\label{thm:program_posterior}
  \rocq{QBS.showcase.bayesian_regression.program_posterior}
  \rocqok
  \uses{def:program, def:posterior, thm:wprior_evidence, lem:qbs_normalize_event, lem:factor_integral, lem:prior_posterior_integral, def:qbs_prob_equiv}
  There exists a probability $q$ on $\mathbb{R} \Rightarrow \mathbb{R}$
  such that $\mathrm{program} = \mathrm{Some}(q)$ and
  $q \sim \mathrm{posterior}$ (up to \texttt{qbs\_prob\_equiv}).
\end{theorem}
\begin{proof}
  \rocqok
  \uses{thm:wprior_evidence, lem:qbs_normalize_event, lem:factor_integral}
  By Theorem~\ref{thm:wprior_evidence} the normaliser succeeds. For
  $U \in \sigma_{M_{\mathbb{R} \Rightarrow \mathbb{R}}}$ apply the event
  formula for the normalised probability to $h = \mathbf{1}_{\mathrm{lin}^{-1}U}$,
  rewrite via Lemma~\ref{lem:factor_integral}, and cancel the evidence~$C$.
\end{proof}
